import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
import os

def generate_dynamic_s7_comparison():
    script_dir = os.path.dirname(os.path.abspath(__file__))
    
    # הגדרת התיקיות החדשות (רמה אחת מעל תיקיית הסקריפט)
    supp_dir = os.path.normpath(os.path.join(script_dir, '..', 'Supplementary_Material'))
    results_dir = os.path.normpath(os.path.join(script_dir, '..', 'preProcessingResults'))

    # 1. טעינת הקבצים שלנו מהתיקייה הייעודית
    our_results_path = os.path.join(results_dir, 'Intra_Study_Results.csv')
    
    # הנחתי שטבלת הגישור נמצאת גם היא בתיקיית התוצאות, אם לא - יש לעדכן את הנתיב
    bridge_table_path = os.path.join(results_dir, 'Study_to_BioProject_Dictionary.csv')
    
    if not os.path.exists(our_results_path) or not os.path.exists(bridge_table_path):
        print("❌ Error: Could not find your results or the bridge table.")
        return

# ... (החלק הראשון של הגדרת הנתיבים נשאר זהה) ...

    df_ours = pd.read_csv(our_results_path)
    df_bridge = pd.read_csv(bridge_table_path)

    # 2. טעינת קובצי S7 מתיקיית ה-Supplementary
    s7_taxon_path = os.path.join(supp_dir, 'Table-S7-AUC_taxon.csv')
    s7_pathway_path = os.path.join(supp_dir, 'Table-S7-AUC_pathway.csv')
    s7_both_path = os.path.join(supp_dir, 'Table-S7-AUC_both.csv')

    try:
        # קריאת שתי העמודות הראשונות (0 ו-1)
        df_taxon = pd.read_csv(s7_taxon_path, usecols=[0, 1])
        df_taxon.columns = ['BioProject_ID', 'Nature_Taxon']
        
        df_pathway = pd.read_csv(s7_pathway_path, usecols=[0, 1])
        df_pathway.columns = ['BioProject_ID', 'Nature_Pathway']
        
        df_both = pd.read_csv(s7_both_path, usecols=[0, 1])
        df_both.columns = ['BioProject_ID', 'Nature_Both']
    except Exception as e:
        print(f"❌ Error reading S7 tables: {e}. Please ensure the files exist and have at least 2 columns.")
        return

    # --- הוספת שלב ניקוי נתונים (Data Cleaning) ---
    print("🧹 Cleaning and standardizing IDs...")
    
    # ניקוי Study_ID
    df_ours['Study_ID'] = df_ours['Study_ID'].astype(str).str.strip().str.upper()
    df_bridge['Study_ID'] = df_bridge['Study_ID'].astype(str).str.strip().str.upper()

    # ניקוי BioProject_ID
    df_bridge['BioProject_ID'] = df_bridge['BioProject_ID'].astype(str).str.strip().str.upper()
    df_taxon['BioProject_ID'] = df_taxon['BioProject_ID'].astype(str).str.strip().str.upper()
    df_pathway['BioProject_ID'] = df_pathway['BioProject_ID'].astype(str).str.strip().str.upper()
    df_both['BioProject_ID'] = df_both['BioProject_ID'].astype(str).str.strip().str.upper()

    # הדפסת דוגמאות לדיבאג
    print("\n🔍 DEBUG - Sample IDs before merge:")
    print(f"Our Studies (first 3): {df_ours['Study_ID'].head(3).tolist()}")
    print(f"Bridge Studies (first 3): {df_bridge['Study_ID'].head(3).tolist()}")
    print(f"Bridge BioProjects (first 3): {df_bridge['BioProject_ID'].head(3).tolist()}")
    print(f"Taxon BioProjects (first 3): {df_taxon['BioProject_ID'].head(3).tolist()}\n")

    # 3. המיזוג הגדול (The Grand Merge)
    print("🔄 Merging local results with Nature S7 tables using the Bridge Table...")
    
    # חיבור התוצאות שלנו למילון הגישור
    df_merged = pd.merge(df_ours, df_bridge, on='Study_ID', how='inner')
    print(f"📊 Rows after merging ours with bridge: {len(df_merged)}")
    
    # חיבור התוצאות של Nature לפי ה-BioProject_ID (בשלבים כדי לראות איפה זה נופל)
    df_merged = pd.merge(df_merged, df_taxon, on='BioProject_ID', how='inner')
    print(f"📊 Rows after merging with Taxon: {len(df_merged)}")
    
    df_merged = pd.merge(df_merged, df_pathway, on='BioProject_ID', how='inner')
    print(f"📊 Rows after merging with Pathway: {len(df_merged)}")
    
    df_merged = pd.merge(df_merged, df_both, on='BioProject_ID', how='inner')
    print(f"📊 Rows after merging with Both: {len(df_merged)}")

    if df_merged.empty:
        print("\n❌ Merge resulted in an empty table. Check the debug outputs above to see where the row count dropped to 0.")
        return

    # ... (המשך הסקריפט המקורי נשאר זהה) ...

    # 4. הכנת הנתונים לגרף
    # אנחנו בוחרים את העמודה של המודל המנצח שלכם (Combined Optuna)
    our_best_method = 'AUC_Combined_Optuna'
    if our_best_method not in df_merged.columns:
        print(f"⚠️ Warning: '{our_best_method}' not found. Using 'AUC_All_Combined' instead.")
        our_best_method = 'AUC_All_Combined'

    methods_to_plot = [our_best_method, 'Nature_Taxon', 'Nature_Pathway', 'Nature_Both']
    
    df_melted = df_merged.melt(id_vars='Study_ID', value_vars=methods_to_plot, 
                               var_name='Method', value_name='AUC')

    # שינוי השמות לתצוגה יפה בגרף
    rename_dict = {
        our_best_method: 'Our Pipeline (Pathways + LogZ + Optuna)',
        'Nature_Taxon': 'Nature Paper (Taxon Only)',
        'Nature_Pathway': 'Nature Paper (Pathway Only)',
        'Nature_Both': 'Nature Paper (Taxon + Pathway)'
    }
    df_melted['Method'] = df_melted['Method'].map(rename_dict)

    # 5. יצירת הגרף ההשוואתי המרובע
    plt.figure(figsize=(16, 8))
    sns.set_theme(style="whitegrid")
    
    # פלטת צבעים: כחול בולט לכם, גווני אדום/כתום לחוקרי Nature
    palette = ['#2980b9', '#e74c3c', '#e67e22', '#8e44ad']
    
    ax = sns.barplot(data=df_melted, x='Study_ID', y='AUC', hue='Method', palette=palette)
    
    plt.axhline(y=0.5, color='black', linestyle='--', alpha=0.5, label='Random Guess (0.5)')
    
    plt.title('Head-to-Head: Our Optimized Pipeline vs. Nature Paper (Table S7)', fontsize=18, fontweight='bold', pad=20)
    plt.xlabel('Dataset (Study ID)', fontsize=14)
    plt.ylabel('Test Set AUC', fontsize=14)
    plt.ylim(0.4, 1.0)
    plt.legend(title='Model Type', bbox_to_anchor=(1.05, 1), loc='upper left')
    
    # הוספת התוויות המדויקות מעל העמודות
    for p in ax.patches:
        height = p.get_height()
        if pd.notnull(height) and height > 0:
            ax.annotate(f'{height:.2f}', 
                        (p.get_x() + p.get_width() / 2., height), 
                        ha='center', va='bottom', 
                        fontsize=9, color='black', xytext=(0, 5), 
                        textcoords='offset points')
    
    plt.tight_layout()
    output_graph = os.path.join(results_dir, 'Nature_S7_Dynamic_Comparison.png')
    plt.savefig(output_graph, dpi=300)
    plt.close()
    
    print(f"✅ Victory Graph successfully created at: {output_graph}")

if __name__ == "__main__":
    generate_dynamic_s7_comparison()
