import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
import os
import glob
import random
import itertools
from math import comb

def load_all_study_pathways(data_folder):
    """
    סורק את תיקיית הדאטה, קורא את קבצי הפיצ'רים, ומחלץ
    עבור כל מחקר את סט התהליכים (Pathways) הייחודיים שלו.
    """
    print("🔍 סורק קבצים ומחלץ תהליכים מכל המחקרים...")
    study_pathways = {}
    
    feature_files = glob.glob(os.path.join(data_folder, "*_features.csv"))
    
    for file_path in feature_files:
        study_name = os.path.basename(file_path).replace("_pathway_abundance_features.csv", "").replace("_features.csv", "")
        
        try:
            df = pd.read_csv(file_path, index_col=0, nrows=1) # קורא רק את הכותרות לחסכון בזיכרון
            pathways = set()
            
            for col in df.columns:
                if '|' in col:
                    pathway = col.split('|', 1)[0]
                    pathways.add(pathway)
            
            if pathways:
                study_pathways[study_name] = pathways
                
        except Exception as e:
            print(f"⚠️ שגיאה בקריאת הקובץ {study_name}: {e}")
            
    print(f"✅ נטענו בהצלחה {len(study_pathways)} מחקרים.")
    return study_pathways

def find_lowest_overlap_pairs(study_pathways, num_pairs_to_remove=1):
    """
    מחשב את החפיפה בין כל זוגות המחקרים האפשריים ומחזיר את הזוגות 
    בעלי החפיפה האחוזית הנמוכה ביותר (ה-Outliers של אושרית).
    """
    studies = list(study_pathways.keys())
    pair_results = []
    
    for study_a, study_b in itertools.combinations(studies, 2):
        intersection = study_pathways[study_a].intersection(study_pathways[study_b])
        union_set = study_pathways[study_a].union(study_pathways[study_b])
        overlap_pct = (len(intersection) / len(union_set)) * 100 if len(union_set) > 0 else 0
        pair_results.append(((study_a, study_b), overlap_pct))
    
    # מיון לפי אחוז החפיפה מהנמוך לגבוה
    pair_results.sort(key=lambda x: x[1])
    
    # שליפת הזוגות הנמוכים ביותר
    lowest_pairs = [pair[0] for pair in pair_results[:num_pairs_to_remove]]
    
    print(f"🚨 זוגות המחקרים עם החפיפה הנמוכה ביותר שנמצאו:")
    for i, (pair, pct) in enumerate(pair_results[:num_pairs_to_remove]):
        print(f"   {i+1}. {pair[0]} ↔️ {pair[1]} ({pct:.2f}% חפיפה)")
        
    return lowest_pairs

def calculate_overlap_data(study_pathways, max_samples_per_k=50, remove_lowest_pairs=False):
    """
    מחשב את חיתוך התהליכים עבור קומבינציות שונות של מחקרים (מ-1 עד N).
    משתמש בדגימה והסרה דינמית מבלי להעמיס על זיכרון ה-RAM.
    """
    studies = list(study_pathways.keys())
    N = len(studies)
    results = []
    
    # איתור הזוגות החלשים במידה והדגל מופעל
    excluded_pairs = []
    if remove_lowest_pairs:
        excluded_pairs = find_lowest_overlap_pairs(study_pathways, num_pairs_to_remove=1) 
    
    for k in range(1, N + 1):
        print(f"⚙️ מחשב חפיפות עבור {k} מחקרים משולבים...")
        
        total_possible_combos = comb(N, k)
        combos_to_process = []
        
        # תרחיש א': מספר הקומבינציות הכולל קטן מסך הדגימות המבוקש (קורה ב-K נמוך או גבוה מאוד)
        if total_possible_combos <= max_samples_per_k:
            all_combos = itertools.combinations(studies, k)
            for combo in all_combos:
                contains_bad_pair = False
                if remove_lowest_pairs:
                    for pair in excluded_pairs:
                        if pair[0] in combo and pair[1] in combo:
                            contains_bad_pair = True
                            break
                if not contains_bad_pair:
                    combos_to_process.append(combo)
                    
        # תרחיש ב': מניעת פיצוץ זיכרון - דגימה אקראית וסינון תוך כדי תנועה (On-the-fly)
        else:
            combos_set = set()
            attempts = 0
            max_attempts = max_samples_per_k * 100 # שומר על הסקריפט מלהיתקע בלולאה אינסופית בערכי K קיצוניים
            
            while len(combos_set) < max_samples_per_k and attempts < max_attempts:
                attempts += 1
                sampled = tuple(sorted(random.sample(studies, k)))
                
                contains_bad_pair = False
                if remove_lowest_pairs:
                    for pair in excluded_pairs:
                        if pair[0] in sampled and pair[1] in sampled:
                            contains_bad_pair = True
                            break
                
                if not contains_bad_pair:
                    combos_set.add(sampled)
            
            combos_to_process = list(combos_set)
            
        # חישוב המדדים לקומבינציות שנבחרו/נדגמו
        for combo in combos_to_process:
            intersection = study_pathways[combo[0]]
            union_set = study_pathways[combo[0]]
            
            for study in combo[1:]:
                intersection = intersection.intersection(study_pathways[study])
                union_set = union_set.union(study_pathways[study])
            
            overlap_count = len(intersection)
            total_unique_in_combo = len(union_set)
            overlap_pct = (overlap_count / total_unique_in_combo) * 100 if total_unique_in_combo > 0 else 0
            
            results.append({
                "Num_Datasets": k,
                "Overlap_Count": overlap_count,
                "Overlap_Percentage": overlap_pct
            })
            
    return pd.DataFrame(results)

def plot_swarm_and_save(df_results, output_dir, file_suffix=""):
    """
    מייצר ושומר שני גרפי Swarm Plot נפרדים בתוך תיקיית הפלט שנבחרה.
    """
    if not os.path.exists(output_dir):
        os.makedirs(output_dir)
        
    N_unique = df_results["Num_Datasets"].nunique()
    
    # --- גרף 1: אבסולוטי ---
    print("📊 מייצר דיאגרמת Swarm Plot אבסולוטית...")
    plt.figure(figsize=(18, 10))
    sns.stripplot(
        data=df_results, x="Num_Datasets", y="Overlap_Count", 
        jitter=0.25, size=4, alpha=0.7, palette="viridis", hue="Num_Datasets", legend=False
    )
    plt.title(f"Pathway Overlap (Absolute Count) Across Combined Datasets {file_suffix.replace('_', ' ').strip()}", fontsize=16, fontweight='bold')
    plt.xlabel("Number of Combined Datasets (k)", fontsize=12)
    plt.ylabel("Number of Common Overlapping Pathways", fontsize=12)
    
    if N_unique > 20:
        step = max(1, N_unique // 20)
        ticks = range(0, N_unique, step)
        plt.xticks(ticks, [str(t + 1) for t in ticks])
        
    plt.grid(axis='y', linestyle='--', alpha=0.5)
    abs_image_path = os.path.join(output_dir, f"pathway_overlap_swarm_absolute{file_suffix}.png")
    plt.savefig(abs_image_path, dpi=300, bbox_inches='tight')
    plt.close()

    # --- גרף 2: אחוזי ---
    print("📊 מייצר דיאגרמת Swarm Plot אחוזית...")
    plt.figure(figsize=(18, 10))
    sns.stripplot(
        data=df_results, x="Num_Datasets", y="Overlap_Percentage", 
        jitter=0.25, size=4, alpha=0.7, palette="plasma", hue="Num_Datasets", legend=False
    )
    plt.title(f"Pathway Overlap (Jaccard Percentage) Across Combined Datasets {file_suffix.replace('_', ' ').strip()}", fontsize=16, fontweight='bold')
    plt.xlabel("Number of Combined Datasets (k)", fontsize=12)
    plt.ylabel("Overlap Percentage of Total Pathways in Combination (%)", fontsize=12)
    plt.ylim(-5, 105)
    
    if N_unique > 20:
        step = max(1, N_unique // 20)
        ticks = range(0, N_unique, step)
        plt.xticks(ticks, [str(t + 1) for t in ticks])
        
    plt.grid(axis='y', linestyle='--', alpha=0.5)
    pct_image_path = os.path.join(output_dir, f"pathway_overlap_swarm_percentage{file_suffix}.png")
    plt.savefig(pct_image_path, dpi=300, bbox_inches='tight')
    plt.close()
    print(f"✅ הגרפים נשמרו בהצלחה בתיקיית היעד עם הסיומת: {file_suffix}")
